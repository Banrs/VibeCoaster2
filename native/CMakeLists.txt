cmake_minimum_required(VERSION 3.20)
project(ExtremeCoaster VERSION 0.8.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(coaster_core core/src/track.cpp core/src/frame.cpp core/src/drive_profile.cpp core/src/simulation.cpp core/src/force_envelope.cpp core/src/convergence.cpp core/src/generation.cpp core/src/layout_modules.cpp core/src/fvd.cpp core/src/persistence.cpp core/src/reference.cpp core/src/supports.cpp core/src/clearance.cpp core/src/dimensions.cpp core/src/station.cpp core/src/structures.cpp)
target_include_directories(coaster_core PUBLIC core/include)
if(MSVC)
  target_compile_options(coaster_core PRIVATE /W4)
  target_compile_options(coaster_core PUBLIC /fp:strict)
else()
  target_compile_options(coaster_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(coaster_core PUBLIC -ffp-contract=off)
endif()
add_executable(coaster_cli core/src/main.cpp)
target_link_libraries(coaster_cli PRIVATE coaster_core)
enable_testing()
find_package(Threads REQUIRED)
add_executable(coaster_tests core/tests/tests.cpp)
target_link_libraries(coaster_tests PRIVATE coaster_core)
add_test(NAME core COMMAND coaster_tests core)
add_test(NAME core_planning COMMAND coaster_tests planning)
add_test(NAME core_station COMMAND coaster_tests station)
add_test(NAME core_placement COMMAND coaster_tests placement)

foreach(suite terrain_transfer terrain_motion passive_transfer terrain_envelope persistence_cancel terrain_baseline circuit_layout reference support support_family layout_modules organic_generation terrain_profile dimensions clearance track_web station structures geometry frame_force terrain convergence drive_profile force_envelope)
  add_executable(${suite}_tests core/tests/${suite}_tests.cpp)
  target_include_directories(${suite}_tests PRIVATE unreal/Source/VibeCoaster/Public)
  target_link_libraries(${suite}_tests PRIVATE coaster_core)
  if(suite STREQUAL "support")
    add_test(NAME ${suite} COMMAND ${suite}_tests "${CMAKE_CURRENT_SOURCE_DIR}/core/tests/fixtures/historical")
  else()
    add_test(NAME ${suite} COMMAND ${suite}_tests)
  endif()
endforeach()
target_link_libraries(persistence_cancel_tests PRIVATE Threads::Threads)
set_tests_properties(persistence_cancel PROPERTIES TIMEOUT 600)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME cli_arguments COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tools/test_cli_arguments.py" "$<TARGET_FILE:coaster_cli>")
  set_tests_properties(cli_arguments PROPERTIES TIMEOUT 180)
endif()

# FVD authoring now supplies production airtime geometry. Keep the public
# target as a compatibility alias without compiling the same object twice.
add_library(coaster_fvd INTERFACE)
target_link_libraries(coaster_fvd INTERFACE coaster_core)
add_executable(fvd_tests core/tests/fvd_tests.cpp)
target_link_libraries(fvd_tests PRIVATE coaster_fvd Threads::Threads)
add_test(NAME fvd COMMAND fvd_tests)

# Explicit matrix convergence tool, excluded from automatic CTest runs.
add_executable(coaster_convergence tools/convergence/audit.cpp)
target_link_libraries(coaster_convergence PRIVATE coaster_core)
