cmake_minimum_required(VERSION 3.24)
project(VibeCoaster2 VERSION 3.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(vibe_core core/src/acceleration.cpp core/src/motion.cpp core/src/track.cpp core/src/spline.cpp core/src/elements.cpp core/src/journey.cpp core/src/recipe.cpp core/src/simulation.cpp core/src/clearance.cpp core/src/terrain.cpp core/src/persistence.cpp)
target_include_directories(vibe_core PUBLIC core/include)
if(MSVC)
  target_compile_options(vibe_core PRIVATE /W4 /permissive- /fp:strict /EHsc)
else()
  target_compile_options(vibe_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(vibe core/src/main.cpp)
target_link_libraries(vibe PRIVATE vibe_core)
enable_testing()
add_executable(acceleration_tests core/tests/acceleration_tests.cpp)
target_link_libraries(acceleration_tests PRIVATE vibe_core)
add_test(NAME acceleration COMMAND acceleration_tests)
add_executable(core_tests core/tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE vibe_core)
add_test(NAME core COMMAND core_tests)


add_executable(vibe_audit core/src/audit.cpp)
target_link_libraries(vibe_audit PRIVATE vibe_core)
add_executable(clearance_tests core/tests/clearance_tests.cpp)
target_link_libraries(clearance_tests PRIVATE vibe_core)
add_test(NAME clearance COMMAND clearance_tests)
add_executable(generator_tests core/tests/generator_tests.cpp)
target_link_libraries(generator_tests PRIVATE vibe_core)
add_test(NAME generator COMMAND generator_tests --extended)

add_executable(persistence_tests core/tests/persistence_tests.cpp)
target_link_libraries(persistence_tests PRIVATE vibe_core)
add_test(NAME persistence COMMAND persistence_tests)
add_executable(terrain_tests core/tests/terrain_tests.cpp)
target_link_libraries(terrain_tests PRIVATE vibe_core)
add_test(NAME terrain COMMAND terrain_tests)
